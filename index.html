/* ========================================================== REDES SOCIALES - SISTEMA CORREGIDO ========================================================== */ const REQUIRED_SOCIAL_TIME = 25000; /* ---------------------------------------------------------- INICIAR UNA RED ---------------------------------------------------------- */ function startSocial(network){ if(!socialState[network]){ return; } if(socialState[network].completed){ return; } const now = Date.now(); socialState[network].openedAt = now; /* Guardamos cada red por separado. Así no se pierde el progreso al salir hacia YouTube, Facebook o WhatsApp. */ localStorage.setItem( "soysigas_social_" + network, String(now) ); } /* ---------------------------------------------------------- COMPROBAR LAS 3 REDES ---------------------------------------------------------- */ function checkSocialTime(){ const networks = [ "youtube", "facebook", "whatsapp" ]; networks.forEach(network => { /* Si ya está completada no hacemos nada. */ if( socialState[network] && socialState[network].completed ){ return; } const savedTime = localStorage.getItem( "soysigas_social_" + network ); if(!savedTime){ return; } const openedAt = Number(savedTime); if(!openedAt){ return; } const elapsed = Date.now() - openedAt; /* Cuando han pasado 25 segundos, se marca automáticamente como completada. */ if(elapsed >= REQUIRED_SOCIAL_TIME){ socialState[network].completed = true; socialState[network].openedAt = null; localStorage.setItem( "soysigas_social_completed_" + network, "1" ); localStorage.removeItem( "soysigas_social_" + network ); markSocialCompleted(network); updateSocialProgress(); } }); /* También recuperamos estados que ya habían sido completados anteriormente. */ networks.forEach(network => { const completed = localStorage.getItem( "soysigas_social_completed_" + network ) === "1"; if(completed){ socialState[network].completed = true; socialState[network].openedAt = null; markSocialCompleted(network); } }); updateSocialProgress(); } /* ---------------------------------------------------------- MARCAR RED COMO COMPLETADA ---------------------------------------------------------- */ function markSocialCompleted(network){ const item = document.getElementById( "social-" + network ); const button = document.getElementById( "button-" + network ); if(!item || !button){ return; } item.classList.add( "completed" ); button.classList.add( "completed" ); button.textContent = "✓ COMPLETADO"; /* Evitamos que vuelva a abrir la red después de completarla. */ button.removeAttribute( "href" ); button.removeAttribute( "target" ); button.removeAttribute( "rel" ); button.onclick = function(event){ event.preventDefault(); return false; }; } /* ---------------------------------------------------------- ACTUALIZAR CONTADOR 0/3, 1/3, 2/3, 3/3 ---------------------------------------------------------- */ function updateSocialProgress(){ const completed = Object.values( socialState ).filter( network => network.completed ).length; const counter = document.getElementById( "socialCount" ); if(counter){ counter.textContent = `${completed} de 3`; } const continueButton = document.getElementById( "continueButton" ); if(!continueButton){ return; } if(completed === 3){ continueButton.classList.add( "active" ); continueButton.textContent = "✅ CONTINUAR"; continueButton.onclick = finishSocialStep; }else{ continueButton.classList.remove( "active" ); continueButton.textContent = "🔒 CONTINUAR"; continueButton.onclick = null; } } /* ---------------------------------------------------------- RECUPERAR ESTADO AL ABRIR LA WEB ---------------------------------------------------------- */ function restoreSocialState(){ const networks = [ "youtube", "facebook", "whatsapp" ]; networks.forEach(network => { const completed = localStorage.getItem( "soysigas_social_completed_" + network ) === "1"; if(completed){ socialState[network].completed = true; socialState[network].openedAt = null; markSocialCompleted(network); return; } const savedTime = localStorage.getItem( "soysigas_social_" + network ); if(savedTime){ socialState[network].openedAt = Number(savedTime); } }); /* Comprobamos inmediatamente si ya pasaron los 25 segundos. */ checkSocialTime(); } /* ---------------------------------------------------------- CUANDO EL USUARIO REGRESA DE OTRA RED ---------------------------------------------------------- */ document.addEventListener( "visibilitychange", function(){ if(!document.hidden){ restoreSocialState(); } } ); /* ---------------------------------------------------------- CUANDO LA PÁGINA VUELVE A ESTAR ACTIVA ---------------------------------------------------------- */ window.addEventListener( "focus", function(){ restoreSocialState(); } ); window.addEventListener( "pageshow", function(){ restoreSocialState(); } );
